cmake_minimum_required(VERSION 3.24)
project(Karizmi LANGUAGES CXX)
add_compile_definitions(NOMINMAX WIN32_LEAN_AND_MEAN _WIN32_WINNT=0x0A00)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
if(MSVC)
  add_compile_options(/utf-8)
endif()
enable_testing()
enable_language(RC)
add_executable(KarizmiStudentService WIN32 src/student_service.cpp service.rc)
target_link_libraries(KarizmiStudentService PRIVATE advapi32 wtsapi32 userenv bcrypt)
add_library(karizmi_core STATIC src/core.cpp)
target_include_directories(karizmi_core PUBLIC include)
if(MSVC)
  target_compile_options(karizmi_core PRIVATE /W4 /EHsc)
endif()
add_executable(karizmi_core_tests tests/core_tests.cpp)
target_link_libraries(karizmi_core_tests PRIVATE karizmi_core)
add_test(NAME core COMMAND karizmi_core_tests)

option(KARIZMI_SECURITY "Build the pinned Botan security component" ON)
set(BOTAN_BUILD_DIR "${CMAKE_CURRENT_SOURCE_DIR}/vendor/botan-build" CACHE PATH "Pinned Botan build directory")
if(KARIZMI_SECURITY)
  if(NOT EXISTS "${BOTAN_BUILD_DIR}/botan-3.lib")
    message(FATAL_ERROR "Build the pinned Botan source first using tools/Build-Core.ps1")
  endif()
  add_library(karizmi_security STATIC src/security.cpp)
  target_include_directories(karizmi_security PRIVATE "${BOTAN_BUILD_DIR}/build/include/public")
  target_link_libraries(karizmi_security PUBLIC karizmi_core PRIVATE "${BOTAN_BUILD_DIR}/botan-3.lib" bcrypt ws2_32 crypt32)
  add_executable(karizmi_security_tests tests/security_tests.cpp)
  target_link_libraries(karizmi_security_tests PRIVATE karizmi_security)
  add_test(NAME security COMMAND karizmi_security_tests)
  add_library(karizmi_network STATIC src/network.cpp src/firewall.cpp src/pairing.cpp src/runtime.cpp src/classroom.cpp src/approvals.cpp)
  target_link_libraries(karizmi_network PUBLIC karizmi_security PRIVATE iphlpapi ole32 oleaut32 advapi32 shell32)
  add_executable(karizmi_network_tests tests/network_tests.cpp)
  target_link_libraries(karizmi_network_tests PRIVATE karizmi_network)
  add_test(NAME network COMMAND karizmi_network_tests)
  set_tests_properties(network PROPERTIES TIMEOUT 30)
  add_library(karizmi_app STATIC src/app.cpp)
  target_link_libraries(karizmi_app PUBLIC karizmi_network PRIVATE comctl32 user32 gdi32 dbghelp)
  add_executable(karizmi_ui_tests tests/ui_tests.cpp)
  target_link_libraries(karizmi_ui_tests PRIVATE karizmi_network comctl32 user32 gdi32 dbghelp shell32)
  add_test(NAME ui_layout COMMAND karizmi_ui_tests)
  set_tests_properties(ui_layout PROPERTIES TIMEOUT 60)
endif()
